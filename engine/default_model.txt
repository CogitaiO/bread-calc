# BreadCalc: модель расчёта (параметры и формулы)
# Файл можно править вручную; порядок формул не важен — программа сама определит порядок вычислений.
# Комментарии при сохранении из программы не сохраняются.
#
# param   ИМЯ = значение | единица | название | note=подсказка
# formula ИМЯ = выражение | единица | название | ref=(5) | group=Группа | final=1

param Wt    = 47.5 | % | Влажность теста W_т | note=для формового ≈ 47.5, для лепёшек ≈ 36–38
param Wm    = 14.5 | % | Влажность муки (базисная)
param GRazd = 0 | % | Разделка/подпыл | note=0.6 — подовый хлеб и лепёшки, 0 — формовой
param GM    = 0.02 | % | Потери муки при приёмке (7)
param GTM   = 0.05 | % | Распыл муки и теста (8)
param GCB   = 3.5 | % | Затраты на брожение (9)
param GUp   = 10 | % | Упёк (11)
param GUs   = 4 | % | Усушка при остывании (12)
param GKr   = 0.03 | % | Крошка и лом (13)
param GSht  = 0.6 | % | Отклонение штучной массы (14)
param GBr   = 0.02 | % | Переработка брака (15)
param GSuh  = 30 | % | Затраты и потери на сухарики | note=норма 27–35

formula Wcp    = SumMW / Mc | % | Средневзвешенная влажность сырья W_ср | ref=(5)
formula Bt     = Mc * (100 - Wcp) / (100 - Wt) | кг | Расчётный выход теста B_т | ref=(4)
formula Pm     = GM * (100 - Wm) / (100 - Wt) | кг | Потери муки при приёмке П_м | ref=(7)
formula Ptm    = GTM * (100 - Wm) / (100 - Wt) | кг | Потери мучной пыли и теста П_тм | ref=(8)
formula Zbr    = (GCB * 0.95 * (Mc - 0.7) * (100 - Wcp)) / (1.96 * 100 * (100 - Wt)) | кг | Затраты сухих веществ на брожение З_бр | ref=(9)
formula Zrazd  = GRazd * (Wt - Wm) / (100 - Wt) | кг | Затраты муки при разделке теста З_разд | ref=(10)
formula Mbake  = Bt - (Pm + Ptm + Zbr + Zrazd) | кг | Масса теста перед посадкой в печь
formula Zup    = GUp * Mbake / 100 | кг | Упёк З_уп | ref=(11)
formula Mhot   = Mbake - Zup | кг | Масса горячего хлеба
formula Zus    = GUs * Mhot / 100 | кг | Усушка при остывании З_ус | ref=(12)
formula Mcool  = Mhot - Zus | кг | Масса остывшего хлеба
formula Pkr    = GKr * Mcool / 100 | кг | Потери крошки и лома П_кр | ref=(13)
formula Mkr    = Mcool - Pkr | кг | Масса после потерь крошки
formula Psht   = GSht * Mkr / 100 | кг | Отклонение штучной массы П_шт | ref=(14)
formula Msht   = Mkr - Psht | кг | Масса после отклонения штучной массы
formula Pbr    = GBr * Msht / 100 | кг | Потери от переработки брака П_бр | ref=(15)
formula Losses = Pm + Ptm + Zbr + Zrazd + Zup + Zus + Pkr + Psht + Pbr | кг | Сумма технологических потерь
formula Bres   = Bt - Losses | кг | ИТОГОВЫЙ ВЫХОД ИЗДЕЛИЯ В_хл^расч | final=1
formula SuhLoss  = Bres * GSuh / 100 | кг | Затраты и потери на сухарики | group=Сухарики
formula SuhYield = Bres - SuhLoss | кг | ИТОГОВЫЙ ВЫХОД СУХАРИКОВ | group=Сухарики | final=1
